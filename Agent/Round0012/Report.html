// eventFilter 捕获点击
bool MainWindow::eventFilter(QObject *obj, QEvent *event) {
    if (obj == m_pathLabel && event->type() == QEvent::MouseButtonPress) {
        onPathLabelClicked(); return true;
    }
    return QMainWindow::eventFilter(obj, event);
}
